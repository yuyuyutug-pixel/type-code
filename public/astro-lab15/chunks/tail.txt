function resetForSourceChange(next){
 if(S.provider===next)return;
 const now=Date.now();
 S.sessionStart=now;
 S.lastForecastAt=now;
 S.model={weights:new Array(10).fill(0),bias:0,n:0,lr:.025,l2:.0004,lastTrainAt:0};
 S.hiddenGaps.push({start:now-16000,end:now+16000});
 for(const f of S.forecasts){
  if(f.status==='pending'){f.status='invalid';f.invalidReason='provider source changed';DB.put('forecasts',f).catch(()=>{})}
 }
 saveLocal();
}

matureForecasts=async function(now=Date.now()){
 const pending=S.forecasts.filter(f=>f.status==='pending'&&now>=f.targetAt);
 for(const r of pending){
  const invalid=r.source!==S.provider||r.createdAt<S.sessionStart||hasGap(r.createdAt,r.targetAt);
  if(invalid){r.status='invalid';r.invalidReason='source switch or timing gap';await DB.put('forecasts',r);continue}
  const tk=S.ticks.find(x=>x.t>=r.targetAt&&x.t<=r.targetAt+1500&&x.s===r.source);
  if(!tk){
   if(now<r.targetAt+1700)continue;
   r.status='invalid';r.invalidReason='no post-expiry tick';await DB.put('forecasts',r);continue
  }
  r.future=tk.p;r.actual=classify(r.entry,r.future);r.status='matured';r.settledAt=tk.t;
  r.pred=r.score>=.5?'UP':'DOWN';
  r.correct=r.actual!=='TIE'&&r.pred===r.actual;
  r.tie=r.actual==='TIE';
  await DB.put('forecasts',r);
  if(r.actual!=='TIE'&&r.quality>=60&&r.createdAt>=(S.model.lastTrainAt||0)+15000){
   S.model.lastTrainAt=r.createdAt;
   updateModel(r.vector,r.actual==='UP'?1:0)
  }
 }
};

providerPerf=function(){
 const data=S.forecasts.filter(x=>x.status==='matured'&&x.source===S.provider).sort((a,b)=>a.createdAt-b.createdAt);
 const a=[];let last=-Infinity;
 for(const r of data){if(r.createdAt>=last+15000){a.push(r);last=r.createdAt}}
 const wins=a.filter(x=>x.correct).length;
 const subset=a.slice(-100);
 return {a,n:a.length,wins,hit:a.length?wins/a.length:0,
  tie:a.length?a.filter(x=>x.tie).length/a.length:0,
  roll100:subset.length?subset.filter(x=>x.correct).length/subset.length:0,
  maxLoss:maxLosing(a.map(x=>!!x.correct)),wilson:wilson(wins,a.length)}
};
